\begin{figure}[htbp]
  \centering
  \includegraphics[width=\linewidth]{Thesis_writeup/Figures/neuroblastoma/Fig_nbl_consensus_dsmz_graph.pdf}
  \caption{{DSMZ neuroblastoma (NBL) cell line consensus similarity network.}
 Nodes represented DSMZ neuroblastoma cell lines.  Edges indicated similarity relationships derived from consensus aggregation of cell line to clinical patient sample neighbourhood analysis. 
 Continuous edges denoted links supported by $\geq 2$ independent feature selection methods (feature-selection and distance metrics).
 Nodes with a thick outline exhibited high betweenness centrality, highlighting potential hub or bridging cell lines. The network was disconnected and comprised two connected components: a component encompassing the majority of NBL cell lines and a distinct two-node component (GI\_ME\_N and CCLF\_PEDS\_0015\_T).}
  \label{fig:nbl_consensus_dsmz_graph}
\end{figure}

\begin{figure}[htbp]
  \centering
  \includegraphics[width=\linewidth]{Figures/breast_cancer/breast_cancer_resolved_components.pdf}
  \caption{{Resolved DSMZ cell line similarity network for breast cancer.}
  Nodes represented DSMZ breast cancer cell lines and edges denoted retained high-confidence similarity links following component resolution of the consensus network. 
  Disconnected components represented distinct transcriptomic groupings. 
  A large connected subgraph spanned luminal/HER2-like cell lines (e.g., \textit{BT\_474}, \textit{MCF\_7}, \textit{T\_47D}, \textit{SK\_BR\_3} and related lines), while a separate basal-like component was centred on \textit{HDQ\_P1} with \textit{CAL\_120}, \textit{CAL\_85\_1}, \textit{HCC\_1143}, and \textit{HCC\_1599}. 
  A smaller linear component connected \textit{JIMT\_1}, \textit{COLO\_824}, and \textit{MDA\_MB\_468}, and additional small or singleton components were observed (e.g., \textit{DU\_4475}, \textit{CAL\_51}, \textit{HCC\_1937}). 
  Black-outlined nodes indicated the representative (anchor) selected for each component, including singleton components.
  }
  \label{fig:brca_resolved_components}
\end{figure}
