\subsection{Supplementary Algorithms}

\begin{algorithm}[H]
\caption{Holistic transcriptomic analysis workflow}
\label{alg:supp_holistic_workflow}
\small
\begin{algorithmic}[1]
\Require DSMZ RNA-seq FASTQ files, clinical tumour RNA-seq data, sample metadata, disease-lineage annotations.
\Ensure Processed expression matrices, feature sets, clusters, tumour neighbourhoods, cell-line graphs, marker sets, and ranking metrics.
\State Collect analysed cell-line RNA-seq profiles and clinical tumour cohorts; per-disease analyses use BRCA, NBL, and RBL, with HEME retained only as an external or excluded comparator where stated.
\State Perform FastQC, STAR two-pass alignment, SAMtools sorting/indexing, RSeQC strandedness checks, Picard metrics, and MultiQC reporting.
\State Estimate tumour purity with ESTIMATE and retain clinical tumours satisfying the study purity threshold.
\State Harmonise gene identifiers, apply lineage-aware ComBat-seq correction, and compute DESeq2 variance-stabilised expression matrices.
\State Generate lineage-specific feature sets, cluster joint tumour--cell-line matrices, and infer cell-line-centred tumour neighbourhoods.
\State Aggregate within-direction and cross-direction consensus evidence, then build patient-referenced cell-line graphs.
\State Resolve stable graph neighbourhoods, connected components, isolates, most-connected nodes, and bridge-like anchors.
\State Select marker genes, recurrence-filter marker pools, construct the pan-cancer feature set, and run enrichment analyses.
\State Build the cell-line-only pan-cancer similarity network, tumour-to-cell-line mapping outputs, and cell-line-to-tumour ranking outputs.
\State Export tables, figures, archived layouts, and reproducibility metadata.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Unsupervised feature selection}
\label{alg:supp_unsupervised_feature_selection}
\small
\begin{algorithmic}[1]
\Require Lineage-specific VST matrix \(X_d\); near-zero SD threshold \(\epsilon=10^{-8}\); first-stage retention \(n_{\mathrm{rank}}=3000\); network-stage retention \(n_{\mathrm{network}}=500\).
\Ensure Nine unsupervised feature sets: Variance, HVG, MAD, MeanAbsDev, Entropy, PCA, Spearman, MX, and kTotal.
\State Remove genes \(g\) with \(\mathrm{SD}(X_{d,g\cdot}) \leq \epsilon\).
\State Compute six first-stage scores: variance; HVG residual log-variance after LOESS mean--variance trend correction; MAD; mean absolute deviation; Shannon entropy using 20 equal-width bins; and summed absolute PCA loading across the first \(\min(5,\mathrm{available\ PCs})\) PCs.
\State For each score, rank genes by decreasing score and break ties by Ensembl gene identifier.
\State Retain the top \(n_{\mathrm{rank}}\) genes per first-stage score.
\State Let \(C_d\) be the union of the six top-\(n_{\mathrm{rank}}\) lists after duplicate removal.
\ForAll{\(g \in C_d\)}
  \State Compute Spearman connectivity \(K_S(g)=|C_d\setminus\{g\}|^{-1}\sum_{h\in C_d,h\neq g}|\rho_S(g,h)|\).
  \State Compute \(\mathrm{ScaledVariance}(g)=\mathrm{Var}(g)/\max_{h\in C_d}\mathrm{Var}(h)\).
  \State Compute the implemented MX score \(\mathrm{MX}(g)=K_S(g)\times\mathrm{ScaledVariance}(g)\).
\EndFor
\State Standardise candidate-gene expression and run WGCNA on \(C_d\) with unsigned adjacency.
\State Select WGCNA soft-threshold power from \(1{:}10,12,14,\ldots,30\) using scale-free fit \(R^2\geq0.8\) and slope in \([-2.5,-1.5]\); use power 5 if no candidate satisfies both criteria.
\State Compute \(kTotal(g)\) as the sum of WGCNA adjacency weights incident on gene \(g\).
\State Rank Spearman connectivity, MX, and \(kTotal\) by decreasing score, using Ensembl gene identifier as the final tie-break.
\State Retain the top \(n_{\mathrm{network}}\) genes for Spearman connectivity, MX, and \(kTotal\).
\State Output the six first-stage sets and three network-refined sets.
\State Record PAM50 separately as a BRCA-only predefined signature, not as an unsupervised feature-selection method.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Clustering and consensus clustering}
\label{alg:supp_clustering_consensus}
\small
\begin{algorithmic}[1]
\Require For lineage \(d\), VST matrix \(X_d\), feature-set collection \(F_d\), distance directions \(\{\mathrm{Euclidean},\mathrm{Pearson}\}\), and grid \(K=\{2,3,4,5,6,7,8\}\) subject to validity caps.
\Ensure Cluster labels per lineage, feature set, distance direction, expression space, and clustering method.
\ForAll{feature sets \(F\in F_d\), including PAM50 only for BRCA}
  \ForAll{distance directions \(q\in\{\mathrm{euc},\mathrm{corr}\}\)}
    \State Build selected-gene expression and top-\(\min(20,\mathrm{available\ PCs})\) PC score matrices.
    \ForAll{expression and PC spaces}
      \State Run hierarchical clustering with Ward D2 using Euclidean distance for \(q=\mathrm{euc}\) and \(1-\rho_{\mathrm{Pearson}}\) for \(q=\mathrm{corr}\).
      \State Select standalone hierarchical \(k\) by maximum mean silhouette width; break ties by smaller \(k\), then representation identifier.
      \If{\(q=\mathrm{euc}\)}
        \State Run k-means with 20 starts on the Euclidean-compatible input.
        \State Select standalone k-means \(k\) by maximum mean silhouette width; break ties by smaller \(k\), then representation identifier.
      \EndIf
      \If{consensus clustering is run for \(q=\mathrm{corr}\)}
        \State Mean-centre each sample vector and divide it by its \(\ell_2\) norm before Euclidean consensus clustering.
      \EndIf
      \State Run ConsensusClusterPlus over valid \(K\) values with 1000 resampling iterations, \(pItem=0.8\), and \(pFeature=1.0\).
      \State Select consensus \(k\) by maximum \(\Delta\)AUC of the consensus CDF; break ties by smaller \(k\), then representation identifier.
    \EndFor
  \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Adaptive cell-line-to-tumour neighbourhood inference}
\label{alg:supp_tumour_neighbourhood_inference}
\small
\begin{algorithmic}[1]
\Require Representation-specific expression matrix, cluster assignments, retained clinical tumours, cell-line samples, distance direction \(q\), and clustering method \(m\).
\Ensure Long-format tumour-neighbourhood table.
\State Harmonise redundant DSMZ profile identifiers after VST/transformation to the canonical profile identifier used by the corresponding analysis, averaging transformed expression values only for identifiers that are redundant at this profile-normalisation step.
\State Map cluster labels to profile-normalised identifiers by majority vote; resolve tied majority votes by cluster label in ascending lexical order.
\State Reserve biological cell-line group collapse for analyses that explicitly use group-level cell-line units, such as the cell-line-centred tumour-class similarity analysis.
\State Restrict expression and cluster assignments to samples present in both objects.
\ForAll{cell lines \(s\)}
  \State Let \(\widetilde{T}(s)\) be retained clinical tumours in the same cluster as \(s\).
  \If{\(|\widetilde{T}(s)| \geq 10\)}
    \State Set \(T(s)=\widetilde{T}(s)\).
  \Else
    \State Set \(T(s)\) to all retained clinical tumours for the lineage.
  \EndIf
  \State Compute Euclidean distance for Euclidean directions, or \(1-\rho_{\mathrm{Pearson}}(s,t)\) for correlation directions.
  \State Sort tumours by increasing distance; break exact ties by tumour identifier.
  \State Set \(h(s)=\min(\max(30,\lceil0.10\,|T(s)|\rceil),200,|T(s)|)\).
  \State Define \(N_{rm}(s)\) as the top \(h(s)\) ranked tumours.
  \State Emit representation, clustering method, cell line, tumour, distance, rank, cell-line cluster, tumour cluster, same-cluster flag, and included flag.
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Within-direction consensus aggregation}
\label{alg:supp_within_direction_consensus}
\small
\begin{algorithmic}[1]
\Require All available neighbourhood tables for one lineage and representation direction \(r\); strong-consensus threshold \(\theta_c=0.7\).
\Ensure Pairwise consensus table with strong-consensus flag or filter, and per-cell-line summary table.
\State Let \(\mathcal{M}_r\) be available contributing clustering methods and set \(M_r=|\mathcal{M}_r|\).
\ForAll{cell-line--tumour pairs \((s,t)\) in the union of neighbourhoods}
  \State \(n_{\mathrm{methods}}(s,t)=|\{m\in\mathcal{M}_r:t\in N_{rm}(s)\}|\).
  \State \(p_{\mathrm{consensus}}(s,t)=n_{\mathrm{methods}}(s,t)/M_r\).
\EndFor
\State Classify strong consensus pairs as \(p_{\mathrm{consensus}}\geq\theta_c\).
\ForAll{cell lines \(s\)}
  \State Summarise maximum consensus, median consensus, fraction with \(p_{\mathrm{consensus}}\geq\theta_c\), and fraction with \(p_{\mathrm{consensus}}=1\) over the tumour union.
\EndFor
\State Output all observed pairwise consensus values, the strong-consensus flag or subset, and per-cell-line summaries sorted by cell-line identifier.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Cross-direction CPI stability summaries}
\label{alg:supp_cross_direction_cpi}
\small
\begin{algorithmic}[1]
\Require Per-direction consensus tables and per-cell-line consensus summaries for a lineage.
\Ensure CPI summaries for cell-line--direction stability and direction-level summaries.
\State Combine valid final consensus tables across directions and discard rows with \(p_{\mathrm{consensus}}\notin[0,1]\).
\ForAll{cell-line--direction pairs \((s,r)\)}
  \State Compute maximum consensus, median consensus, fraction with \(p_{\mathrm{consensus}}\geq0.7\), and fraction with \(p_{\mathrm{consensus}}=1\).
\EndFor
\State Compute direction-level summaries from per-cell-line metrics so that each cell line contributes equally.
\State Remove metrics with standard deviation \(\leq10^{-12}\).
\State Standardise each remaining metric to mean 0 and standard deviation 1.
\State Run PCA on the standardised metric matrix.
\State Set \(\omega_k=|\mathrm{loading}_{k,1}|/\sum_j|\mathrm{loading}_{j,1}|\) for each retained metric.
\State Compute \(\mathrm{CPI}_{sr}=\sum_k\omega_k z_{srk}\).
\State Rank directions per cell line by decreasing CPI, then decreasing fraction \(\geq0.7\), median consensus, maximum consensus, and representation name.
\State Use CPI as a descriptive summary of cross-representation neighbourhood stability; CPI is not used as an input feature, weight, or ranking criterion for the final correlation-based tumour-to-cell-line mapping.
\State Output PCA-derived metric weights, cell-line--direction CPI scores, best directions per cell line, and direction-level summaries.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Representation-specific patient-referenced cell-line similarity graph construction}
\label{alg:cell_line_similarity_network}
\small
\begin{algorithmic}[1]
\Require Consensus probabilities \(p_{\mathrm{consensus}}(p,t)\), cell-line profiles \(P\), profile-to-biological-group map \(g(p)\), retained biological groups \(S\), retained clinical tumours \(T=\{t_1,\ldots,t_P\}\), and threshold quantile \(q=0.90\).
\Ensure Undirected weighted patient-referenced cell-line similarity graph \(G_r\).
\ForAll{cell-line profiles \(p\)}
  \State Represent \(p\) as \(v_p=(p_{\mathrm{consensus}}(p,t_1),\ldots,p_{\mathrm{consensus}}(p,t_P))\), filling absent tumour pairs with 0.
\EndFor
\ForAll{biological groups \(s\in S\)}
  \State Set \(v_s\) to the arithmetic mean of all \(v_p\) with \(g(p)=s\); groups represented by one profile retain that profile vector unchanged.
\EndFor
\State Compute pairwise Pearson correlations \(\mathrm{sim}(s_i,s_j)=\rho_{\mathrm{Pearson}}(v_{s_i},v_{s_j})\).
\State Construct symmetric similarity matrix \(M\), set the diagonal to 1, and exclude diagonal entries from threshold estimation.
\State Let \(\tau\) be the empirical \(q\)-quantile of off-diagonal similarities.
\State Build an undirected weighted graph with nodes equal to biological cell-line groups only, not tumours.
\State Add edge \((s_i,s_j)\) if \(i<j\) and \(M_{ij}\geq\tau\); assign edge weight \(M_{ij}\).
\ForAll{nodes \(s\)}
  \State Compute degree and isolate flag \(\mathbb{I}[\deg(s)=0]\).
\EndFor
\State Run Leiden and Louvain community detection as graph annotations.
\State Output similarity matrix, edge table, node summary, isolate table, and community annotations.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Support-threshold consensus network across representations}
\label{alg:supp_support_threshold_consensus_network}
\small
\begin{algorithmic}[1]
\Require Representation-specific biological-group graph edge sets
\(\{E_r:r\in\mathcal{R}\}\), support threshold \(m\).
\Ensure Support-threshold consensus network \(G_{\mathrm{support}}(m)\).
\ForAll{unordered biological-group pairs \(e=(s_i,s_j)\)}
  \State Compute support count
  \(a(e)=|\{r\in\mathcal{R}:e\in E_r\}|\).
  \State Compute support fraction
  \(f(e)=a(e)/|\mathcal{R}|\).
\EndFor
\State Retain edge \(e\) if \(a(e)\geq m\).
\State Construct \(G_{\mathrm{support}}(m)\) from retained edges.
\State Annotate retained edges with support count \(a(e)\) and support fraction \(f(e)\).
\State Output the support-threshold consensus edge table and graph summary.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Detailed graph-based consensus resolution}
\label{alg:supp_graph_based_consensus_resolution}
\small
\begin{algorithmic}[1]
\Require Representation-specific patient-referenced biological-group graphs \(G_r=(S,E_r)\), representation-level summaries, and per-group summaries.
\Ensure Resolved graph \(G_{\mathrm{final}}\), edge table, node summary, components, isolates, most-connected nodes, and bridge-like anchors.
\State Select global best representation \(r^\ast\) by decreasing fraction above consensus threshold, median consensus, and mean consensus; break remaining ties by representation name.
\ForAll{biological groups \(i\)}
  \State Select local best representation set \(T_i\) by the implemented per-group primary criterion, fraction above the consensus threshold.
  \State Retain all exact co-optimal local ties in \(T_i\).
  \State Set \(N_{\mathrm{global}}(i)=N_{r^\ast}(i)\).
  \State Set \(N_{\mathrm{local}}(i)=\bigcup_{r\in T_i}N_r(i)\).
  \State Set \(N_{\mathrm{final}}(i)=N_{\mathrm{global}}(i)\cap N_{\mathrm{local}}(i)\).
\EndFor
\State Construct \(G_{\mathrm{final}}\) as an undirected graph from stable-neighbour pairs over the biological-group node set.
\State Decompose \(G_{\mathrm{final}}\) into connected components and retain degree-zero nodes as isolates.
\ForAll{non-isolate connected components}
  \State Compute node degree and unnormalised betweenness on unweighted shortest paths.
  \State Select the most-connected node by maximum degree, breaking ties by biological-group identifier.
  \State Select the bridge-like anchor by maximum unnormalised betweenness, breaking ties by biological-group identifier.
\EndFor
\State Output resolved edges, node statistics, component membership, isolates, most-connected nodes, and bridge-like anchors.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Marker-gene selection, recurrence filtering, and pan-cancer feature construction}
\label{alg:supp_marker_recurrence_pan_cancer_features}
\small
\begin{algorithmic}[1]
\Require Raw count matrix, resolved graph per disease, isolate set, bridge-like anchor set, and component membership.
\Ensure Per-contrast marker tables, strict recurrent markers, isolate-rescue markers, and final pan-cancer feature set.
\ForAll{disease lineages \(d\)}
  \State Define isolate-vs-REST contrasts with target equal to one isolate cell line and REST equal to all other cell lines in \(d\).
  \State Define bridge-like-anchor-vs-REST contrasts with target equal to one bridge-like anchor and REST equal to all cell lines outside the anchor's resolved component.
  \State Run DESeq2 on raw counts for each contrast.
  \State Retain isolate markers with adjusted \(p\leq0.01\), \(\mathrm{baseMean}\geq10\), \(|\log_2\mathrm{FC}|\geq1.5\), and at least 10 normalised counts in the target.
  \State Retain anchor markers with adjusted \(p\leq0.05\), \(\mathrm{baseMean}\geq10\), \(|\log_2\mathrm{FC}|\geq1.0\), and at least 10 normalised counts in the target.
  \State Rank markers by descending \(|\log_2\mathrm{FC}|\), ascending adjusted \(p\)-value, and Ensembl gene identifier.
  \State Truncate isolate marker lists to top 50 genes and anchor marker lists to top 200 genes.
  \State Retain strict recurrent genes appearing in at least two contrasts within \(d\).
  \State Rank non-recurrent isolate rescue candidates by contrast support descending, best adjusted \(p\)-value ascending, maximum \(|\log_2\mathrm{FC}|\) descending, best marker-list rank ascending, and Ensembl gene identifier ascending.
  \State Add up to 50 isolate-rescue genes for \(d\).
\EndFor
\State Merge disease-level marker pools, deduplicate by Ensembl gene identifier, and verify final size; the size is 257 genes in the present cohort.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Pan-cancer cell-line transcriptomic similarity network}
\label{alg:supp_pan_cancer_cell_line_network}
\small
\begin{algorithmic}[1]
\Require DSMZ RNA-seq profiles, final pan-cancer feature set, \(k_{\mathrm{NN}}=20\), lineage metadata, and archived layout coordinates.
\Ensure Cell-line-only pan-cancer graph and validation summaries.
\State Restrict RNA-seq profiles to the final pan-cancer feature set.
\State Compute pairwise Spearman correlations and use \(1-\rho_S\) for nearest-neighbour ordering.
\ForAll{RNA-seq profiles \(s\)}
  \State Select \(k_{\mathrm{NN}}\) nearest neighbours by decreasing Spearman correlation, excluding \(s\); break ties by profile identifier.
\EndFor
\State Construct a union kNN graph by adding an undirected edge if either endpoint selected the other.
\State Weight edges by Spearman correlation.
\State Use fixed archived layout coordinates for analysis figures.
\State Distinguish this cell-line-only graph from the patient-referenced graph in Supplementary Algorithm~\ref{alg:cell_line_similarity_network}.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Tumour-to-cell-line transcriptomic mapping and classification}
\label{alg:supp_tumour_to_cell_line_mapping}
\small
\begin{algorithmic}[1]
\Require Retained clinical tumours, cell lines, final pan-cancer feature set, and lineage labels.
\Ensure Per-tumour rankings, predicted-lineage table, and cohort-level metrics.
\State Restrict clinical tumour and cell-line matrices to pan-cancer feature genes.
\State Apply cohort-wise gene \(z\)-standardisation; in the final configured run, skip the optional broad global-program filter because no gene-symbol mapping is supplied.
\ForAll{tumours \(t\)}
  \ForAll{cell lines \(c\)}
    \State Compute Spearman similarity \(s(t,c)\).
  \EndFor
  \State Rank cell lines by decreasing similarity; use average tied ranks for rank summaries and cell-line identifier for deterministic top-hit selection.
  \State Predict tumour lineage as the lineage of the top-ranked cell line.
  \State Compute confidence delta \(s(t,c_{(1)})-s(t,c_{(10')})\), where \(10'=\min(10,\mathrm{number\ of\ cell\ lines})\).
  \State Compute top-10 lineage purity.
\EndFor
\State Compute Acc@1, Acc@10, MRR, confidence-delta summaries, top-k lineage-purity summaries, and non-focal lineage ECDF/reference distributions.
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Cell-line-to-tumour similarity ranking}
\label{alg:supp_cellline_tumour_class_similarity}
\label{alg:supp_cell_line_centred_tumour_retrieval}
\small
\begin{algorithmic}[1]
\Require Purity-filtered clinical tumours, biological cell-line groups, final pan-cancer feature set, patient consensus components, and reciprocal depth \(k=10\).
\Ensure Cell-line ranking metrics, reciprocal pair table, and component-composition table.
\State Group technical cell-line profiles into biological cell-line groups.
\ForAll{biological cell-line groups \(s\)}
  \If{\(s\) is RBL\_15 or RBL\_20}
    \State Compute per-library tumour Spearman similarities and collapse each tumour score by median across libraries.
  \Else
    \State Use the single profile or group-level expression profile for \(s\).
  \EndIf
  \State Rank all clinical tumours by decreasing Spearman similarity; break ties by tumour identifier.
  \State Compare the top-ranked tumour lineage with the cell-line lineage.
  \State Compute same-lineage median rank percentile, top-10\% matching fraction, rank1--rank2 margin, and rank1--rank10 margin.
\EndFor
\State Compute top-ranked lineage accuracy, balanced accuracy, and Wilson confidence intervals.
\State Identify reciprocal tumour--cell-line pairs using top-\(k\) depth \(k=10\).
\State Map top 50 tumours per cell line to patient consensus network components and summarise component composition.
\end{algorithmic}
\end{algorithm}
