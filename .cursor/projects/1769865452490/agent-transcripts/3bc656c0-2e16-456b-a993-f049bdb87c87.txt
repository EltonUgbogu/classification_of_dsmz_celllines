```python
#!/usr/bin/env snakemake
# =============================================================================
# RNA-seq preprocessing pipeline (paired-end)
#
# Scientific scope:
#   - Read-level QC using FastQC
#   - Splice-aware alignment to a reference genome using STAR (2-pass)
#   - Gene-level count extraction from STAR GeneCounts output
#   - Alignment-level QC using Picard metrics
#   - Cohort-level QC aggregation using MultiQC
#
# Primary inputs:
#   - config/samples.tsv  (sample, r1, r2; r1/r2 may be comma-separated for multi-lane)
#   - config/config.yaml  (STAR index, reference FASTA, Picard jar, threads)
#
# Output layout (consistent across all steps):
#   results/fastqc/
#   results/star/
#   results/star/counts/
#   results/picard/
#   results/multiqc/
#   logs/
# =============================================================================

import os
import pandas as pd

# -----------------------------------------------------------------------------
# Configuration
# -----------------------------------------------------------------------------
configfile: "config/config.yaml"

THREADS    = int(config.get("threads", 12))
STAR_INDEX = config["genome"]["star_index"]
REF_FASTA  = config["genome"]["fasta"]
GTF        = config["genome"].get("gtf", "")          # Optional: kept for completeness
PICARD_JAR = config["tools"]["picard_jar"]

# -----------------------------------------------------------------------------
# Sample sheet (paired-end; supports multi-lane)
# -----------------------------------------------------------------------------
SAMPLES_DF = pd.read_csv("config/samples.tsv", sep="\t")
if not {"sample", "r1", "r2"}.issubset(set(SAMPLES_DF.columns)):
    raise ValueError("config/samples.tsv must contain columns: sample, r1, r2")

SAMPLES = dict(sorted(SAMPLES_DF.set_index("sample").T.to_dict().items()))
SAMPLE_IDS = list(SAMPLES.keys())

# -----------------------------------------------------------------------------
# Helpers: parse lanes, validate existence, produce STAR-ready strings
# -----------------------------------------------------------------------------
def _split_csv_paths(x):
    return [p.strip() for p in str(x).split(",") if p.strip()]

def _require_files_exist(sample, paths, label):
    missing = [p for p in paths if not os.path.exists(p)]
    if missing:
        raise FileNotFoundError(
            f"[INPUT] Missing {label} FASTQ file(s) for sample '{sample}':\n  " + "\n  ".join(missing)
        )

def get_r1_star(wc):
    r1_list = _split_csv_paths(SAMPLES[wc.sample]["r1"])
    _require_files_exist(wc.sample, r1_list, "R1")
    # STAR accepts comma-separated lanes for each mate
    return ",".join(r1_list)

def get_r2_star(wc):
    r2_list = _split_csv_paths(SAMPLES[wc.sample]["r2"])
    _require_files_exist(wc.sample, r2_list, "R2")
    return ",".join(r2_list)

def get_all_fastqs(wc):
    r1_list = _split_csv_paths(SAMPLES[wc.sample]["r1"])
    r2_list = _split_csv_paths(SAMPLES[wc.sample]["r2"])
    _require_files_exist(wc.sample, r1_list, "R1")
    _require_files_exist(wc.sample, r2_list, "R2")
    return r1_list + r2_list

# -----------------------------------------------------------------------------
# Final targets
# -----------------------------------------------------------------------------
rule all:
    input:
        # STAR alignment + indexing + counts
        expand("results/star/{sample}.Aligned.sortedByCoord.out.bam", sample=SAMPLE_IDS),
        expand("results/star/{sample}.Aligned.sortedByCoord.out.bam.bai", sample=SAMPLE_IDS),
        expand("results/star/counts/{sample}.ReadsPerGene.out.tab", sample=SAMPLE_IDS),
        expand("results/star/{sample}.Log.final.out", sample=SAMPLE_IDS),

        # FastQC completion markers (handles multi-lane cleanly)
        expand("results/fastqc/{sample}.fastqc.done", sample=SAMPLE_IDS),

        # Picard metrics
        expand("results/picard/{sample}_validate_samfile.txt", sample=SAMPLE_IDS),
        expand("results/picard/{sample}_alignment_summary_metrics.txt", sample=SAMPLE_IDS),
        expand("results/picard/{sample}_quality_yield_metrics.txt", sample=SAMPLE_IDS),
        expand("results/picard/{sample}_insert_size_metrics.txt", sample=SAMPLE_IDS),
        expand("results/picard/{sample}_insert_size_histogram.pdf", sample=SAMPLE_IDS),

        # MultiQC
        "results/multiqc/multiqc_report.html"

# -----------------------------------------------------------------------------
# FastQC
# -----------------------------------------------------------------------------
# FastQC produces one HTML+ZIP per FASTQ input. For multi-lane samples, that means
# multiple outputs. A sentinel file is therefore used as the stable “per-sample”
# output that downstream steps can depend on.
rule fastqc:
    input:
        fq = get_all_fastqs
    output:
        done = "results/fastqc/{sample}.fastqc.done"
    log:
        "logs/fastqc/{sample}.log"
    conda:
        "envs/fastqc.yaml"
    threads: 2
    shell:
        r"""
        set -euo pipefail
        mkdir -p results/fastqc logs/fastqc

        # Run FastQC on all FASTQ files (R1 and R2; supports multi-lane)
        fastqc --threads {threads} --outdir results/fastqc {input.fq} > {log} 2>&1

        # Sentinel marks successful completion for the sample
        touch {output.done}
        """

# -----------------------------------------------------------------------------
# STAR 2-pass alignment + GeneCounts
# -----------------------------------------------------------------------------
rule star_2pass:
    input:
        index = STAR_INDEX,
        r1 = get_r1_star,
        r2 = get_r2_star,
        fq_done = "results/fastqc/{sample}.fastqc.done"
    output:
        bam    = "results/star/{sample}.Aligned.sortedByCoord.out.bam",
        bai    = "results/star/{sample}.Aligned.sortedByCoord.out.bam.bai",
        counts = "results/star/counts/{sample}.ReadsPerGene.out.tab",
        logf   = "results/star/{sample}.Log.final.out"
    log:
        "logs/star/{sample}.log"
    benchmark:
        "logs/benchmarks/star/{sample}.tsv"
    conda:
        "envs/star.yaml"
    threads: THREADS
    resources:
        mem_mb = 42000
    params:
        prefix = lambda wc: f"results/star/{wc.sample}."
    shell:
        r"""
        set -euo pipefail
        mkdir -p results/star results/star/counts logs/star logs/benchmarks/star

        # STAR alignment (2-pass improves splice junction discovery in RNA-seq)
        STAR --genomeDir "{input.index}" \
             --readFilesIn "{input.r1}" "{input.r2}" \
             --readFilesCommand zcat \
             --outFileNamePrefix "{params.prefix}" \
             --outSAMtype BAM SortedByCoordinate \
             --quantMode GeneCounts \
             --twopassMode Basic \
             --runThreadN {threads} \
             --limitBAMsortRAM 32000000000 \
             > "{log}" 2>&1

        # Index BAM for downstream QC and rapid access
        samtools index "{output.bam}"

        # Move the gene count table into a stable location
        mv -f "{params.prefix}ReadsPerGene.out.tab" "{output.counts}"
        """

# -----------------------------------------------------------------------------
# Picard QC (alignment-level metrics)
# -----------------------------------------------------------------------------
rule picard_validate_samfile:
    input:
        bam = rules.star_2pass.output.bam,
        ref = REF_FASTA
    output:
        "results/picard/{sample}_validate_samfile.txt"
    log:
        "logs/picard/{sample}_validate_samfile.log"
    conda:
        "envs/picard.yaml"
    shell:
        r"""
        set -euo pipefail
        mkdir -p results/picard logs/picard
        java -Xmx8g -jar "{PICARD_JAR}" ValidateSamFile \
            I="{input.bam}" O="{output}" MODE=SUMMARY REFERENCE_SEQUENCE="{input.ref}" \
            > "{log}" 2>&1 || true
        """

rule picard_alignment_summary_metrics:
    input:
        bam = rules.star_2pass.output.bam,
        ref = REF_FASTA
    output:
        "results/picard/{sample}_alignment_summary_metrics.txt"
    log:
        "logs/picard/{sample}_alignment_summary_metrics.log"
    conda:
        "envs/picard.yaml"
    shell:
        r"""
        set -euo pipefail
        mkdir -p results/picard logs/picard
        java -jar "{PICARD_JAR}" CollectAlignmentSummaryMetrics \
            R="{input.ref}" I="{input.bam}" O="{output}" > "{log}" 2>&1
        """

rule picard_quality_yield_metrics:
    input:
        bam = rules.star_2pass.output.bam
    output:
        "results/picard/{sample}_quality_yield_metrics.txt"
    log:
        "logs/picard/{sample}_quality_yield_metrics.log"
    conda:
        "envs/picard.yaml"
    shell:
        r"""
        set -euo pipefail
        mkdir -p results/picard logs/picard
        java -jar "{PICARD_JAR}" CollectQualityYieldMetrics \
            I="{input.bam}" O="{output}" > "{log}" 2>&1
        """

rule picard_insert_size_metrics:
    input:
        bam = rules.star_2pass.output.bam,
        ref = REF_FASTA
    output:
        metrics = "results/picard/{sample}_insert_size_metrics.txt",
        hist    = "results/picard/{sample}_insert_size_histogram.pdf"
    log:
        "logs/picard/{sample}_insert_size_metrics.log"
    conda:
        "envs/picard.yaml"
    shell:
        r"""
        set -euo pipefail
        mkdir -p results/picard logs/picard
        java -jar "{PICARD_JAR}" CollectInsertSizeMetrics \
            I="{input.bam}" O="{output.metrics}" H="{output.hist}" \
            M=0.5 R="{input.ref}" VALIDATION_STRINGENCY=LENIENT > "{log}" 2>&1
        """

# -----------------------------------------------------------------------------
# MultiQC (aggregate FastQC + STAR + Picard)
# -----------------------------------------------------------------------------
rule multiqc:
    input:
        expand("results/fastqc/{sample}.fastqc.done", sample=SAMPLE_IDS),
        expand("results/star/{sample}.Log.final.out", sample=SAMPLE_IDS),
        expand("results/picard/{sample}_insert_size_metrics.txt", sample=SAMPLE_IDS),
        expand("results/picard/{sample}_alignment_summary_metrics.txt", sample=SAMPLE_IDS),
        expand("results/picard/{sample}_quality_yield_metrics.txt", sample=SAMPLE_IDS),
        expand("results/picard/{sample}_validate_samfile.txt", sample=SAMPLE_IDS)
    output:
        html = "results/multiqc/multiqc_report.html"
    log:
        "logs/multiqc/multiqc.log"
    conda:
        "envs/multiqc.yaml"
    threads: 2
    shell:
        r"""
        set -euo pipefail
        mkdir -p results/multiqc logs/multiqc

        # MultiQC scans the results tree and aggregates supported outputs
        multiqc results -o results/multiqc -n multiqc_report.html > "{log}" 2>&1
        """
```
