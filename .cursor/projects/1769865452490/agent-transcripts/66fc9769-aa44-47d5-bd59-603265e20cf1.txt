```r
clean_ensg_rows <- function(mat) {
  rownames(mat) <- sub("\\.\\d+$", "", rownames(mat))
  mat
}

load_matrix <- function(path) {
  if (!file.exists(path)) stop("Missing file: ", path)
  m <- readRDS(path)
  m <- as.matrix(m)
  m
}

make_dataset_meta <- function(cfg) {
  tibble(
    id   = names(cfg),
    path = unname(cfg)
  ) %>%
    mutate(
      cancer_type = case_when(
        grepl("nbl",  id, ignore.case=TRUE) ~ "NBL",
        grepl("brca|bcc", id, ignore.case=TRUE) ~ "BRCA",
        grepl("rbl",  id, ignore.case=TRUE) ~ "RBL",
        grepl("leu_lym|laml", id, ignore.case=TRUE) ~ "LEU/LYM",
        TRUE ~ "OTHER"
      ),
      sample_type = case_when(
        grepl("dsmz", id, ignore.case=TRUE) ~ "Cell Line",
        TRUE ~ "Tumour"
      )
    )
}

merge_by_common_genes <- function(dataset_meta) {
  mats <- purrr::map(dataset_meta$path, load_matrix)
  names(mats) <- dataset_meta$id
  mats <- purrr::map(mats, clean_ensg_rows)

  common_genes <- Reduce(intersect, purrr::map(mats, rownames))
  cat("[INFO] Common genes across datasets:", length(common_genes), "\n")

  mats <- purrr::map(mats, ~ .x[common_genes, , drop=FALSE])
  merged <- do.call(cbind, mats)  # genes x samples

  annot <- purrr::imap_dfr(mats, function(mat, nm) {
    tibble(sample = colnames(mat), dataset_id = nm)
  }) %>%
    left_join(dataset_meta, by = c("dataset_id" = "id"))

  annot <- annot[match(colnames(merged), annot$sample), ]
  stopifnot(all(annot$sample == colnames(merged)))

  list(merged = merged, annot = annot, common_genes = common_genes)
}

run_umap_block <- function(X_sxg, annot, source_label) {
  summaries <- list()

  for (subset_mode in c("TUMOUR_ONLY", "INTEGRATED")) {

    if (subset_mode == "TUMOUR_ONLY") {
      keep <- annot$sample_type == "Tumour"
      X_use <- X_sxg[keep, , drop = FALSE]
      annot_use <- annot[keep, , drop = FALSE]
    } else {
      X_use <- X_sxg
      annot_use <- annot
    }

    for (dm in dist_metrics) {

      tag <- paste(source_label, subset_mode, dm, sep = "_")
      cat("\n[INFO] Running UMAP:", tag, "\n")
      set.seed(opt$seed)

      emb <- uwot::umap(
        X           = X_use,
        n_neighbors = opt$n_neighbors,
        min_dist    = opt$min_dist,
        metric      = dm,
        n_threads   = n_threads,
        scale       = FALSE,
        verbose     = TRUE
      )

      coords <- as.data.frame(emb)
      colnames(coords) <- c("UMAP1","UMAP2")
      coords$sample_id <- rownames(X_use)

      coords <- coords %>%
        left_join(
          annot_use %>% transmute(sample_id = sample, cancer_type, sample_type, dataset_id),
          by = "sample_id"
        )

      coords_path <- file.path(outdir, paste0("coords_", tag, ".tsv"))
      write_tsv(coords, coords_path)

      plot_df <- prepare_plot_data(coords)

      subtitle <- sprintf(
        "%s; %s; metric=%s; nn=%d; min_dist=%.2f; genes=%d",
        source_label,
        ifelse(subset_mode == "TUMOUR_ONLY", "tumours only", "tumours + cell lines"),
        dm, opt$n_neighbors, opt$min_dist, ncol(X_use)
      )

      if (subset_mode == "TUMOUR_ONLY") {
        p <- plot_tumour_only(plot_df %>% filter(type == "Tumour"),
                              disease_palette, subtitle)
      } else {
        p <- plot_main_overview(plot_df, disease_palette, subtitle)
      }

      plot_base <- file.path(outdir, paste0("Fig_umap_", tag))
      save_plot(p, plot_base, page = opt$page)

      summaries[[tag]] <- tibble(
        source      = source_label,
        subset      = tolower(subset_mode),
        dist_metric = dm,
        n_genes     = ncol(X_use),
        n_samples   = nrow(X_use),
        n_tumours   = sum(annot_use$sample_type == "Tumour"),
        n_cell_lines= sum(annot_use$sample_type == "Cell Line"),
        coords_tsv  = basename(coords_path),
        plot_pdf    = basename(paste0(plot_base, ".pdf")),
        plot_png    = basename(paste0(plot_base, ".png"))
      )

      cat("[SAVED] ", coords_path, "\n", sep="")
      cat("[SAVED] ", paste0(plot_base, ".pdf/.png"), "\n", sep="")
    }
  }

  bind_rows(summaries)
}
```

clean_ensg_rows <- function(mat) {
  rownames(mat) <- sub("\\.\\d+$", "", rownames(mat))
  mat
}

load_matrix <- function(path) {
  if (!file.exists(path)) stop("Missing file: ", path)
  m <- readRDS(path)
  m <- as.matrix(m)
  m
}

make_dataset_meta <- function(cfg) {
  tibble(
    id   = names(cfg),
    path = unname(cfg)
  ) %>%
    mutate(
      cancer_type = case_when(
        grepl("nbl",  id, ignore.case=TRUE) ~ "NBL",
        grepl("brca|bcc", id, ignore.case=TRUE) ~ "BRCA",
        grepl("rbl",  id, ignore.case=TRUE) ~ "RBL",
        grepl("leu_lym|laml", id, ignore.case=TRUE) ~ "LEU/LYM",
        TRUE ~ "OTHER"
      ),
      sample_type = case_when(
        grepl("dsmz", id, ignore.case=TRUE) ~ "Cell Line",
        TRUE ~ "Tumour"
      )
    )
}

merge_by_common_genes <- function(dataset_meta) {
  mats <- purrr::map(dataset_meta$path, load_matrix)
  names(mats) <- dataset_meta$id
  mats <- purrr::map(mats, clean_ensg_rows)

  common_genes <- Reduce(intersect, purrr::map(mats, rownames))
  cat("[INFO] Common genes across datasets:", length(common_genes), "\n")

  mats <- purrr::map(mats, ~ .x[common_genes, , drop=FALSE])
  merged <- do.call(cbind, mats)  # genes x samples

  annot <- purrr::imap_dfr(mats, function(mat, nm) {
    tibble(sample = colnames(mat), dataset_id = nm)
  }) %>%
    left_join(dataset_meta, by = c("dataset_id" = "id"))

  annot <- annot[match(colnames(merged), annot$sample), ]
  stopifnot(all(annot$sample == colnames(merged)))

  list(merged = merged, annot = annot, common_genes = common_genes)
}

# SECTION 7: UMAP RUNNER (RAW/BATCH × TUMOUR_ONLY/INTEGRATED)
# ==============================================================================
run_umap_block <- function(X_sxg, annot, source_label) {
  summaries <- list()
  pt_size   <- opt$point_size
  pt_alpha  <- opt$point_alpha_tumour

  for (subset_mode in c("TUMOUR_ONLY", "INTEGRATED")) {

    if (subset_mode == "TUMOUR_ONLY") {
      keep <- annot$sample_type == "Tumour"
      X_use <- X_sxg[keep, , drop = FALSE]
      annot_use <- annot[keep, , drop = FALSE]
    } else {
      X_use <- X_sxg
      annot_use <- annot
    }

    for (dm in dist_metrics) {

      tag <- paste(source_label, subset_mode, dm, sep = "_")
      cat("\n[INFO] Running UMAP:", tag, "\n")
      set.seed(opt$seed)

      emb <- uwot::umap(
        X           = X_use,
        n_neighbors = opt$n_neighbors,
        min_dist    = opt$min_dist,
        metric      = dm,
        n_threads   = n_threads,
        scale       = FALSE,
        verbose     = TRUE
      )

      coords <- as.data.frame(emb)
      colnames(coords) <- c("UMAP1","UMAP2")
      coords$sample_id <- rownames(X_use)

      coords <- coords %>%
        left_join(
          annot_use %>% transmute(sample_id = sample, cancer_type, sample_type, dataset_id),
          by = "sample_id"
        )

      coords_path <- file.path(outdir, paste0("coords_", tag, ".tsv"))
      write_tsv(coords, coords_path)

      plot_df <- prepare_plot_data(coords)

      subtitle <- sprintf(
        "%s; %s; metric=%s; nn=%d; min_dist=%.2f; genes=%d",
        source_label,
        ifelse(subset_mode == "TUMOUR_ONLY", "tumours only", "tumours + cell lines"),
        dm, opt$n_neighbors, opt$min_dist, ncol(X_use)
      )

      if (subset_mode == "TUMOUR_ONLY") {
        p <- plot_tumour_only(plot_df %>% filter(type == "Tumour"),
                              disease_palette, subtitle, point_size = pt_size, point_alpha_tumour = pt_alpha)
      } else {
        p <- plot_main_overview(plot_df, disease_palette, subtitle, point_size = pt_size, point_alpha_tumour = pt_alpha)
      }

      plot_base <- file.path(outdir, paste0("Fig_umap_", tag))
      save_plot(p, plot_base, page = opt$page)

      summaries[[tag]] <- tibble(
        source      = source_label,
        subset      = tolower(subset_mode),
        dist_metric = dm,
        n_genes     = ncol(X_use),
        n_samples   = nrow(X_use),
        n_tumours   = sum(annot_use$sample_type == "Tumour"),
        n_cell_lines= sum(annot_use$sample_type == "Cell Line"),
        coords_tsv  = basename(coords_path),
        plot_pdf    = basename(paste0(plot_base, ".pdf")),
        plot_png    = basename(paste0(plot_base, ".png"))
      )

      cat("[SAVED] ", coords_path, "\n", sep="")
      cat("[SAVED] ", paste0(plot_base, ".pdf/.png"), "\n", sep="")
    }
  }

  bind_rows(summaries)
}

```r
plot_tumour_only <- function(df, palette, subtitle_text, caption_text,
                             label_side = c("right","left"),

      plot.title  = element_text(face = "bold"),

      subtitle = subtitle_text,
      caption  = caption_text
    )

  # Side labels: anchor at extreme x of each cluster + median y
  label_df <- df %>%
    group_by(disease) %>%

    size = label_size,
    label.size = 0.15,

  )
}
```

```r
plot_integrated_celligner <- function(df, palette, subtitle_text) {

  # Celligner naming: type = tumour vs CL
  df2 <- df %>%
    mutate(
      type = ifelse(sample_type == "Cell Line", "CL", "tumour"),
      type = factor(type, levels = c("tumour", "CL"))
    )

  ggplot(df2, aes(UMAP1, UMAP2)) +

    # Tumours first (white outline)
    geom_point(
      data  = df2 %>% filter(type == "tumour"),
      aes(fill = disease, colour = type, size = type),
      shape = 21,
      alpha = 0.70,
      stroke = 0.45
    ) +

    # Cell lines on top (black outline, slightly thicker)
    geom_point(
      data  = df2 %>% filter(type == "CL"),
      aes(fill = disease, colour = type, size = type),
      shape = 21,
      alpha = 0.70,
      stroke = 0.70
    ) +

    # Fill = cancer type
    scale_fill_manual(values = palette, name = "Cancer type") +

    # Outline = sample type (Celligner trick)
    scale_colour_manual(values = c("CL" = "black", "tumour" = "white"), guide = "none") +

    # Size = sample type (Celligner trick)
    scale_size_manual(values = c("CL" = 1.10, "tumour" = 0.85), guide = "none") +

    theme_classic(base_size = 11) +
    theme(
      legend.position = "bottom",
      legend.direction = "horizontal",
      legend.title = element_text(face = "plain", size = 9),
      legend.text  = element_text(size = 8),
      legend.margin = margin(0, 0, 0, 0)
    ) +
    guides(fill = guide_legend(override.aes = list(shape = 21, size = 3, alpha = 1))) +
    labs(
      title = NULL,
      subtitle = subtitle_text,
      x = "UMAP 1",
      y = "UMAP 2"
    )
}
```

### Why this matches Celligner

* `shape = 21` + `fill = disease` is the “colour blob” look.
* `colour = type` with **tumour = white**, **CL = black** gives the crisp ring effect.
* `size = type` makes CLs pop without needing a different shape.
* Drawing tumours first and CLs second is essential.
